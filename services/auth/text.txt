auth/better-auth.ts(import type { Database } from '@uhs/service-kit';
import { betterAuth } from 'better-auth';
import { drizzleAdapter } from 'better-auth/adapters/drizzle';
import { jwt, organization, twoFactor } from 'better-auth/plugins';
import type { ServiceConfig } from '../config.js';

/**
 * Better Auth owns accounts, sessions, MFA, organisations (tenants) and
 * memberships (spec §5). Other services never import this: they verify the
 * issued JWTs against the JWKS.
 *
 * The auth tables are generated with `npm run db:auth-generate` once the
 * Better Auth schema is finalised; they live in this service's database only.
 */
export function createAuth(config: ServiceConfig, database: Database) {
  return betterAuth({
    appName: 'UHS',
    baseURL: config.BETTER_AUTH_URL,
    secret: config.BETTER_AUTH_SECRET,
    database: drizzleAdapter(database.db, { provider: 'pg' }),
    emailAndPassword: {
      enabled: true,
      requireEmailVerification: true,
    },
    session: {
      expiresIn: 60 * 60 * 24 * 7,
      updateAge: 60 * 60,
    },
    plugins: [
      // A tenant is a Better Auth organisation; a facility is a team.
      organization(),
      // Issues the access token described in spec §5.5.
      jwt(),
      // MFA is mandatory for staff memberships.
      twoFactor(),
    ],
    advanced: {
      database: { generateId: false },
    },
  });
}

export type Auth = ReturnType<typeof createAuth>;

/**
 * Unified identity payload for all microservices.
 * Other services call this (or trust JWT) to know what the user can do.
 */
export async function getAccess(req: Request, res: Response) {
  try {
    const user = req.user!;
    const role = (user.role || "patient").toLowerCase();
    const permissions = await PermissionModel.findByRole(role);
    const keys = permissions.map((p) => p.key);

    // Group by service for portals
    const byService: Record<string, string[]> = {};
    for (const p of permissions) {
      const sk = p.serviceKey || "auth";
      if (!byService[sk]) byService[sk] = [];
      byService[sk].push(p.key);
    }

    const staffProfile =
      role !== "patient" ? await StaffModel.findByUserId(user.id) : null;
    const patientProfile =
      role === "patient" ? await PatientModel.findByUserId(user.id) : null;

    res.json({
      user: {
        id: user.id,
        name: user.name,
        email: user.email,
        role,
      },
      permissions: keys,
      services: byService,
      staffProfile,
      patientProfile,
    });
  } catch (e) {
    console.error(e);
    res.status(500).json({ error: "Failed to resolve access" });
  }
}

export async function checkPermission(req: Request, res: Response) {
  try {
    const needed = String(req.query.permission || req.body?.permission || "");
    if (!needed) {
      return res.status(400).json({ error: "permission query/body required" });
    }
    const role = (req.user?.role || "patient").toLowerCase();
    const permissions = await PermissionModel.findByRole(role);
    const allowed = permissions.some((p) => p.key === needed);
    res.json({ permission: needed, allowed, role });
  } catch (e) {
    console.error(e);
    res.status(500).json({ error: "Failed to check permission" });
  }
}

src/middleware/auth.ts(import type { Request, Response, NextFunction } from "express";
import { auth } from "../auth.js";

export type SessionUser = {
  id: string;
  name: string;
  email: string;
  role?: string | null;
  [key: string]: unknown;
};

declare global {
  namespace Express {
    interface Request {
      user?: SessionUser;
      session?: unknown;
    }
  }
}

export async function requireAuth(
  req: Request,
  res: Response,
  next: NextFunction
) {
  try {
    const session = await auth.api.getSession({
      headers: req.headers as any,
    });
    if (!session?.user) {
      return res.status(401).json({ error: "Unauthorized" });
    }
    req.user = session.user as SessionUser;
    req.session = session.session;
    next();
  } catch {
    return res.status(401).json({ error: "Unauthorized" });
  }
}

export function requireRoles(...roles: string[]) {
  return (req: Request, res: Response, next: NextFunction) => {
    const role = (req.user?.role || "").toLowerCase();
    if (!roles.map((r) => r.toLowerCase()).includes(role)) {
      return res.status(403).json({
        error: "Forbidden",
        message: `Requires one of: ${roles.join(", ")}`,
      });
    }
    next();
  };
}

  "pharmacist",

src/middleware/validate.ts(import type { Request, Response, NextFunction } from "express";
import type { ZodSchema, ZodError } from "zod";

type Target = "body" | "query" | "params";

/**
 * Validate req.body / req.query / req.params against a Zod schema.
 * On failure returns 400 with field-level errors.
 */
export function validate(schema: ZodSchema, target: Target = "body") {
  return (req: Request, res: Response, next: NextFunction) => {
    const result = schema.safeParse(req[target]);

    if (!result.success) {
      const error = result.error as ZodError;
      return res.status(400).json({
        error: "Validation failed",
        details: error.errors.map((e) => ({
          path: e.path.join("."),
          message: e.message,
        })),
      });
    }

    // Replace with parsed (coerced / stripped) data
    req[target] = result.data;
    next();
  };
}
